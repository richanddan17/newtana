using UnityEngine;

/// <summary>
/// 보스 전용 AI - 패턴 기반, 페이즈 전환, 특수 능력.
/// EnemyAI를 상속받아 보스 전용 로직 추가.
/// 계획서 Phase 8, enemy-ai-system.md [8] 확장.
///
/// [수정 내역]
/// 1. UnityEditor 코드를 #if UNITY_EDITOR로 감싸 빌드 오류 해결
/// 2. 리플렉션(GetField) 제거 -> 기본 속도를 Awake에서 캐싱 (속도 0 버그 해결)
/// 3. 최대 체력 버프 제거 (페이즈 연쇄 전환 / 배율 누적 버그 해결)
/// 4. 분노 모드, 데미지 배율, 쉴드 흡수, 미니언 소환 연결
/// 5. 쉴드 이벤트 호출 빈도 제한, 체력 0 나누기 방어
/// </summary>
[DisallowMultipleComponent]
public class BossAI : EnemyAI
{
    [Header("Boss Specific")]
    [SerializeField] float[] phaseHealthThresholds = { 0.66f, 0.33f }; // 체력 비율로 페이즈 전환
    [SerializeField] float speedBonusPerPhase = 0.15f;   // 페이즈당 속도 증가율
    [SerializeField] float damageBonusPerPhase = 0.25f;  // 페이즈당 데미지 증가율
    [SerializeField] bool enrageAtLowHealth = true;
    [SerializeField] float enrageHealthRatio = 0.2f;
    [SerializeField] float enrageSpeedMultiplier = 1.5f;
    [SerializeField] float enrageDamageMultiplier = 1.5f;

    [Header("Special Abilities")]
    [SerializeField] bool hasShield = false;
    [SerializeField] float shieldRechargeTime = 10f;     // 마지막 피격 후 재충전 시작까지 대기 시간
    [SerializeField] float maxShield = 50f;
    [SerializeField] float shieldRechargePercentPerSec = 0.1f; // 초당 재충전 비율 (최대 쉴드 대비)

    [Header("Arena")]
    [SerializeField] Transform[] spawnPoints; // 소환 위치
    [SerializeField] GameObject minionPrefab; // 소환할 미니언
    [SerializeField] int minionsPerPhase = 2; // 페이즈 전환 시 소환 수 (0이면 소환 안 함)

    // 상태
    int currentPhaseIndex = 0;
    float currentShield;
    bool isEnraged = false;
    float shieldRechargeTimer;
    float lastNotifiedShield; // 이벤트 스팸 방지용

    // 기본값 캐싱 (배율이 누적되지 않도록 항상 이 값 기준으로 계산)
    float baseMoveSpeed;
    bool baseSpeedCached = false;

    // 이벤트
    public event System.Action<int> OnPhaseChanged; // newPhaseIndex
    public event System.Action OnEnrageActivated;
    public event System.Action<float, float> OnShieldChanged; // current, max

    protected override void Awake()
    {
        base.Awake();
        enemyType = EnemyAIData.EnemyType.Boss;
        currentShield = maxShield;
        lastNotifiedShield = currentShield;
        shieldRechargeTimer = 0f;

        CacheBaseStats();
    }

    void CacheBaseStats()
    {
        if (movement == null) return;

        // TODO: 이동 클래스에 기본 추격 속도를 읽는 프로퍼티가 필요합니다.
        // 예) public float BaseChaseSpeed => chaseSpeed;
        baseMoveSpeed = movement.BaseChaseSpeed;
        baseSpeedCached = true;
    }

    protected override void Update()
    {
        base.Update();

        if (health == null || health.IsDead) return;

        UpdateShield();
        CheckPhaseTransition();
        CheckEnrage();
    }

    // ───────────────────────── 쉴드 ─────────────────────────

    void UpdateShield()
    {
        if (!hasShield || currentShield >= maxShield) return;

        // 피격 후 대기 시간 동안은 재충전하지 않음
        if (shieldRechargeTimer > 0f)
        {
            shieldRechargeTimer -= Time.deltaTime;
            return;
        }

        currentShield = Mathf.Min(maxShield, currentShield + maxShield * shieldRechargePercentPerSec * Time.deltaTime);
        NotifyShieldChanged(false);
    }

    void NotifyShieldChanged(bool force)
    {
        // 최대 쉴드의 1% 이상 변했을 때만 알림 (UI 스팸 방지)
        if (force || Mathf.Abs(currentShield - lastNotifiedShield) >= maxShield * 0.01f || currentShield >= maxShield)
        {
            lastNotifiedShield = currentShield;
            OnShieldChanged?.Invoke(currentShield, maxShield);
        }
    }

    /// <summary>
    /// 쉴드가 먼저 데미지를 흡수하고, 남은 데미지를 반환합니다.
    /// EnemyHealth.TakeDamage 맨 앞에서 호출하세요:
    ///   if (boss != null) damage = boss.AbsorbDamage(damage);
    ///   if (damage <= 0f) return;
    /// </summary>
    public float AbsorbDamage(float damage)
    {
        if (!hasShield || currentShield <= 0f || damage <= 0f) return damage;

        float absorbed = Mathf.Min(currentShield, damage);
        currentShield -= absorbed;
        shieldRechargeTimer = shieldRechargeTime;
        NotifyShieldChanged(true);

        // 쉴드 파괴 시 잠시 스턴
        if (currentShield <= 0f && health != null)
            health.ForceStagger(1f);

        return damage - absorbed; // 쉴드를 뚫고 들어가는 남은 데미지
    }

    // ───────────────────────── 페이즈 ─────────────────────────

    float HealthRatio
    {
        get
        {
            if (health == null || health.MaxHealth <= 0f) return 1f;
            return health.CurrentHealth / health.MaxHealth;
        }
    }

    void CheckPhaseTransition()
    {
        // while: 한 번에 큰 데미지를 받아 여러 페이즈를 건너뛰어도 순서대로 처리
        while (currentPhaseIndex < phaseHealthThresholds.Length &&
               HealthRatio <= phaseHealthThresholds[currentPhaseIndex])
        {
            AdvancePhase();
        }
    }

    void AdvancePhase()
    {
        currentPhaseIndex++;
        currentPhase = currentPhaseIndex; // EnemyAI의 currentPhase 동기화 (0 기반인지 EnemyAI에서 확인 필요)

        OnPhaseChanged?.Invoke(currentPhaseIndex);

        ApplyStats();

        if (minionsPerPhase > 0)
            SpawnMinions(minionsPerPhase);

        // 패턴 쿨다운 리셋으로 즉시 새 패턴 선택
        patternCooldownTimer = 0f;
        currentPatternIndex = -1;

        Debug.Log($"[BossAI] Phase {currentPhaseIndex + 1} started!");
    }

    // ───────────────────────── 능력치 (페이즈 + 분노) ─────────────────────────

    /// <summary>현재 페이즈/분노 상태에 따른 이동 속도 배율. (항상 기본값 기준이라 누적되지 않음)</summary>
    public float SpeedMultiplier =>
        (1f + currentPhaseIndex * speedBonusPerPhase) * (isEnraged ? enrageSpeedMultiplier : 1f);

    /// <summary>
    /// 현재 페이즈/분노 상태에 따른 데미지 배율.
    /// 무기/공격 코드에서 최종 데미지에 곱해서 사용하세요:
    ///   float dmg = baseDamage * boss.DamageMultiplier;
    /// </summary>
    public float DamageMultiplier =>
        (1f + currentPhaseIndex * damageBonusPerPhase) * (isEnraged ? enrageDamageMultiplier : 1f);

    void ApplyStats()
    {
        if (movement != null && baseSpeedCached)
            movement.SetMoveSpeed(baseMoveSpeed * SpeedMultiplier);
    }

    // ───────────────────────── 분노 ─────────────────────────

    void CheckEnrage()
    {
        if (isEnraged || !enrageAtLowHealth) return;

        if (HealthRatio <= enrageHealthRatio)
            ActivateEnrage();
    }

    void ActivateEnrage()
    {
        isEnraged = true;
        ApplyStats(); // 분노 배율 반영
        OnEnrageActivated?.Invoke();
        Debug.Log("[BossAI] ENRAGE ACTIVATED!");
    }

    // ───────────────────────── 기타 ─────────────────────────

    public int CurrentPhase => currentPhaseIndex + 1;
    public int TotalPhases => (phaseHealthThresholds != null ? phaseHealthThresholds.Length : 0) + 1;
    public float ShieldRatio => maxShield > 0f ? currentShield / maxShield : 0f;
    public bool IsEnraged => isEnraged;

    // 소환 능력
    public void SpawnMinions(int count)
    {
        if (minionPrefab == null || spawnPoints == null || spawnPoints.Length == 0) return;

        for (int i = 0; i < count; i++)
        {
            var spawnPoint = spawnPoints[Random.Range(0, spawnPoints.Length)];
            if (spawnPoint == null) continue;
            Instantiate(minionPrefab, spawnPoint.position, spawnPoint.rotation);
        }
    }

    protected override void OnDied()
    {
        base.OnDied();
        // 보스 사망 이펙트, 드롭 등
    }

#if UNITY_EDITOR
    protected override void OnDrawGizmosSelected()
    {
        base.OnDrawGizmosSelected();

        // 페이즈 전환 체력 표시
        if (phaseHealthThresholds != null)
        {
            for (int i = 0; i < phaseHealthThresholds.Length; i++)
            {
                UnityEditor.Handles.Label(
                    transform.position + Vector3.up * (3f + i * 0.5f),
                    $"Phase {i + 2} at {phaseHealthThresholds[i] * 100:F0}% HP"
                );
            }
        }

        if (hasShield)
        {
            UnityEditor.Handles.Label(
                transform.position + Vector3.up * 4f,
                $"Shield: {currentShield:F0}/{maxShield:F0}"
            );
        }
    }
#endif
}
